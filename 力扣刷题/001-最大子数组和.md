## 💻 代码实现（C++）

class Solution {

public:

    int maxSubArray(vector<int>& nums) {

        int n = nums.size();

        if (n == 0) return 0;

  

        vector<int>D(n);

        D[0] = nums[0];

        int max_sum = D[0];


        for (int i = 1; i < n; i++)

        {

            D[i] = max(D[i - 1] + nums[i], nums[i]);

            max_sum  = max(max_sum, D[i]);

        }

        return max_sum;

  

    }

};
