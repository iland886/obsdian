- 解法二（优化）：优化的关键点在遍历到nums[i]时，如果nums[i] - 1 在去重后的nums中，说明nums[i]不是第一个连续的数字，直接跳过，然后一直遍历nums[i] ++, 同时更新序列长度。
- 核心算法思想：动态规划 / 贪心 / 哈希表 / 双指针 等

## 💻 代码实现（C++）

```cpp
class Solution {

public:

    int longestConsecutive(vector<int>& nums) {

        unordered_set<int> num_set;

        for (const int& num : nums)

        {

            num_set.insert(num);

        }

        int longestStreak  = 0;

  
  

        for (int num : num_set)

        {

            if (!num_set.count(num - 1))

            {

                int currentNum = num;

                int currentStreak = 1;

  

                while (num_set.count(currentNum + 1))

                {

                    currentNum += 1;

                    currentStreak += 1;

                }

  

                longestStreak = max (longestStreak, currentStreak);

            }

        }

        return longestStreak;

    }

};
